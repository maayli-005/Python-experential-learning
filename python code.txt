# Student Marksheet Automation

students = [
    {"roll": 101, "name": "Aarav",
     "marks": [78, 85, 91, 88, 80]},

    {"roll": 102, "name": "Diya",
     "marks": [92, 89, 95, 90, 94]}
]

for s in students:
    total = sum(s["marks"])
    percentage = total / (len(s["marks"]) * 100) * 100

    if percentage >= 90:
        grade = "A+"
    elif percentage >= 80:
        grade = "A"
    elif percentage >= 70:
        grade = "B"
    elif percentage >= 60:
        grade = "C"
    else:
        grade = "D"

    print(s["roll"], s["name"],
          total, round(percentage, 2), grade)
